```typescript
export type TestCase = {
  ID: string;                       // "TC-{PREFIX}-{NNN}"
  Назва: string;                    // short name
  Передумови: string;               // preconditions
  Кроки: string[];                  // ordered steps (non-empty array)
  "Очікуваний результат": string;   // expected result
  Пріоритет: "High" | "Medium" | "Low";
};
```
